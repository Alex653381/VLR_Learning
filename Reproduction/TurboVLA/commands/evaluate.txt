# TurboVLA LIBERO 评测命令

cd ~/workspace1/VLR_Learning/TurboVLA

source ~/miniconda3/etc/profile.d/conda.sh
conda activate turbovla-libero

export HF_ENDPOINT=https://hf-mirror.com
export TORCH_FORCE_NO_WEIGHTS_ONLY_LOAD=1

python experiments/libero/evaluate.py \
  --ckpt_path pretrained/TurboVLA/checkpoints/libero/turbovla_libero_eval.pth \
  --libero_root /home/alex/workspace1/LIBERO \
  --dinov3_path pretrained/dinov3-vitb16-standin \
  --bert_path pretrained/bert-base-uncased \
  --stats_path experiments/libero/configs/libero_all4_stats.json \
  --stats_key libero_all4_no_noops \
  --task_suite_name libero_spatial \
  --num_trials_per_task 50 \
  --chunk_size 12 \
  --num_open_loop_steps 12 \
  --seed 7 \
  --precision bf16 \
  --result_json_path outputs/evaluation/libero_spatial_50.json
